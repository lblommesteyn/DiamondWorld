set -euo pipefail

export CUDA_VISIBLE_DEVICES=0
export XLA_PYTHON_CLIENT_PREALLOCATE=false

BASE_TAG="abcd_hbp_e_s42"
FT_TAG="${BASE_TAG}_ft15"

mkdir -p data/eval2

.venv/bin/python -m diamondworldjax.scripts.train_pitchformer \
  --stack abcd \
  --steps 6000 \
  --bs 32 \
  --seed 42 \
  --tag "${BASE_TAG}" \
  > "data/train_pitchformer_${BASE_TAG}.log" 2>&1

.venv/bin/python -m diamondworldjax.scripts.finetune_pitchformer_rollout \
  --init-tag "${BASE_TAG}" \
  --tag "${FT_TAG}" \
  --steps 2000 \
  --batch 32 \
  --lr 1e-5 \
  --self-conditioned-fraction 0.15 \
  --ss-warmup 500 \
  --seed 42 \
  > "data/finetune_pitchformer_${FT_TAG}.log" 2>&1

.venv/bin/python -m diamondworldjax.scripts.eval_pitchformer_rollout \
  --tag "${FT_TAG}" \
  --season 2024 \
  --reps 3 \
  --seed 42 \
  --batch 512 \
  --c-events \
  --out "data/eval2/pitchformer_${FT_TAG}_correlations.json" \
  > "data/eval_pitchformer_${FT_TAG}_correlations.log" 2>&1

.venv/bin/python -m diamondworldjax.scripts.eval_pitchformer_games \
  --tag "${FT_TAG}" \
  --season 2024 \
  --seed 42 \
  --batch-games 512 \
  --c-events \
  --out "data/eval2/pitchformer_${FT_TAG}_games.json" \
  > "data/eval_pitchformer_${FT_TAG}_games.log" 2>&1

  CUDA_VISIBLE_DEVICES=1 .venv/bin/python scripts/run_ladder.sh \
  v22pf_s42 0 --pitchformer --skip-train --steps 50000 --sim-replicas 100 --sim-chunk 250
